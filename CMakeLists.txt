cmake_minimum_required(VERSION 3.24)
project(seat_service VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SEAT_SANITIZERS "Enable ASan and UBSan" OFF)
find_package(SQLite3 3.54.0 EXACT REQUIRED)
set(OPENSSL_ROOT_DIR "/opt/homebrew/opt/openssl@3" CACHE PATH "OpenSSL prefix")
find_package(OpenSSL 3.6.3 EXACT REQUIRED)
add_library(seat_core src/domain.cpp src/service.cpp src/database.cpp src/password.cpp)
if(TARGET SQLite3::SQLite3)
 target_link_libraries(seat_core PUBLIC SQLite3::SQLite3 OpenSSL::Crypto)
else()
 target_link_libraries(seat_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto)
endif()
find_package(Threads REQUIRED)
target_include_directories(seat_core PUBLIC include)
target_compile_options(seat_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(SEAT_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
 target_compile_options(seat_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(seat_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(seat_cli main.cpp)
target_link_libraries(seat_cli PRIVATE seat_core)
enable_testing()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/doctest.h" doctest_sha)
if(NOT doctest_sha STREQUAL "94029a7d32da24a56249658147dbd2b33ff0b9ed665295cbbaf19aafff5b0ced")
 message(FATAL_ERROR "Vendored doctest 2.4.12 hash mismatch")
endif()
add_executable(seat_tests tests/core_test.cpp)
target_include_directories(seat_tests SYSTEM PRIVATE third_party)
target_link_libraries(seat_tests PRIVATE seat_core Threads::Threads)
add_test(NAME core COMMAND seat_tests)
option(SEAT_HTTP "Build Drogon HTTP service" OFF)
if(SEAT_HTTP)
 include(cmake/HttpDependencies.cmake)
 add_executable(seat_http src/http.cpp)
 target_link_libraries(seat_http PRIVATE seat_core drogon)
 target_compile_options(seat_http PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/cli_smoke.py --cli $<TARGET_FILE:seat_cli>)
set_tests_properties(core PROPERTIES TIMEOUT 60)
set_tests_properties(cli PROPERTIES TIMEOUT 20)
if(SEAT_HTTP)
 add_test(NAME http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/http_smoke.py --build $<TARGET_FILE_DIR:seat_http>)
 set_tests_properties(http PROPERTIES TIMEOUT 90)
endif()
add_executable(seat_crash_writer tests/crash_writer.cpp)
target_link_libraries(seat_crash_writer PRIVATE seat_core)
add_test(NAME crash_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/crash_smoke.py --build $<TARGET_FILE_DIR:seat_crash_writer>)
set_tests_properties(crash_recovery PROPERTIES TIMEOUT 20)

foreach(project_target seat_cli seat_tests seat_crash_writer)
 target_compile_options(${project_target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
